procedure TForm1.MenuItemOpenImageClick(Sender: TObject);
var
  SrcBitmap, DestBitmap: TBitmap;
begin
  OpenDialog1.Filter := 'BMP Images (*.bmp)|*.bmp';

  if OpenDialog1.Execute then
  begin
    OriginalFilePath := OpenDialog1.FileName;
    WorkingBMPPath := ExtractFilePath(ParamStr(0)) + 'working.bmp';

    SrcBitmap := TBitmap.Create;
    DestBitmap := TBitmap.Create;
    try
      SrcBitmap.LoadFromFile(OriginalFilePath);

      // Force the destination to be a completely clean 24-bit container
      DestBitmap.PixelFormat := pf24bit;
      DestBitmap.Width := SrcBitmap.Width;
      DestBitmap.Height := SrcBitmap.Height;

      // Draw the old image perfectly onto the new 24-bit canvas
      DestBitmap.Canvas.Draw(0, 0, SrcBitmap);

      // Save the clean 24-bit copy
      DestBitmap.SaveToFile(WorkingBMPPath);

      // Display it cleanly
      Image1.Picture.LoadFromFile(WorkingBMPPath);

      if CheckBoxFit.Checked then
         CheckBoxFitChange(Sender);

    finally
      SrcBitmap.Free;
      DestBitmap.Free;
    end;
  end;
end;



procedure TForm1.MenuItemResetClick(Sender: TObject);
begin
  if WorkingBMPPath <> '' then
    Image1.Picture.LoadFromFile(WorkingBMPPath);
end;

procedure TForm1.MenuItemSaveImageClick(Sender: TObject);
var
TargetFile: String;
SourceStream, DestStream: TFileStream;
BMP_DataBend_Path: String;
begin
  if OriginalFilePath = '' then
  begin
    ShowMessage('There is no image to save yet.');
    Exit;
  end;

  BMP_DataBend_Path := ExtractFilePath(OriginalFilePath) + 'new.bmp';


  if not FileExists(BMP_DataBend_Path) then
  begin
    ShowMessage('Error, no filepath.');
    Exit;
  end;

  // save file parameters
  SaveDialog1.Title := 'Save Image As';
  SaveDialog1.Filter := 'BMP Images (*.bmp)|*.bmp';
  SaveDialog1.DefaultExt := 'bmp';


  if SaveDialog1.Execute then
  begin
    TargetFile := SaveDialog1.FileName;

    SourceStream := TFileStream.Create(BMP_DataBend_Path, fmOpenRead);
    try
      DestStream := TFileStream.Create(TargetFile, fmCreate);
      try
        DestStream.CopyFrom(SourceStream, SourceStream.Size);
      finally
        DestStream.Free;
      end;
    finally
      SourceStream.Free;
    end;

    ShowMessage('Saved successfully!');
  end;
end;

procedure TForm1.Button1Click(Sender: TObject);
var
  InStream, OutStream: TFileStream;
  HeaderBuffer: array of Byte; // Changed to dynamic array
  PixelBuffer: array of Byte;

  PixelDataOffset: LongWord;   // 32-bit unsigned integer for the offset
  PixelDataSize: Int64;
  OutputPath: String;
  Intensity: Integer;
begin
  if OpenDialog1.FileName = '' then
  begin
    ShowMessage('Please load a BMP image from the File menu.');
    Exit;
  end;

  Intensity := TrackBar1.Position;
  OutputPath := ExtractFilePath(OriginalFilePath) + 'new.bmp';

  InStream := TFileStream.Create(WorkingBMPPath, fmOpenRead);
  OutStream := TFileStream.Create(OutputPath, fmCreate);

  try
    // 1. Read the first 14 bytes to find out where the header ends
    SetLength(HeaderBuffer, 14);
    InStream.Read(HeaderBuffer[0], 14);

    // 2. Extract the 32-bit integer from bytes 10-13
    // Move maps the 4 raw bytes directly into our LongWord variable
    Move(HeaderBuffer[10], PixelDataOffset, 4);

    // 3. Resize our header buffer to match the actual header size and read the rest
    SetLength(HeaderBuffer, PixelDataOffset);
    InStream.Position := 0; // Reset stream pointer to beginning
    InStream.Read(HeaderBuffer[0], PixelDataOffset);

    // 4. Write the exact header block to the output file
    OutStream.Write(HeaderBuffer[0], PixelDataOffset);

    // 5. Calculate the remaining raw pixel bytes safely
    PixelDataSize := InStream.Size - PixelDataOffset;
    SetLength(PixelBuffer, PixelDataSize);
    InStream.Read(PixelBuffer[0], PixelDataSize);

    // Run your data bending functions
    case RadioGroup1.ItemIndex of
      0: ApplyChannelSwap(PixelBuffer, PixelDataSize, Intensity);
      1: ApplyInvertBlock(PixelBuffer, PixelDataSize, Intensity);
      2: ApplyEchoSmear(PixelBuffer, PixelDataSize, Intensity);
      3: ApplyVaporwavePosterizer(PixelBuffer, PixelDataSize, Intensity);
      4: ApplyCRTScanlines(PixelBuffer, PixelDataSize, Intensity);
    end;

    // Write modified pixels
    OutStream.Write(PixelBuffer[0], PixelDataSize);

  finally
    SetLength(HeaderBuffer, 0);
    SetLength(PixelBuffer, 0);
    InStream.Free;
    OutStream.Free;
  end;

  // UI Refresh logic
  Image1.Picture.Assign(nil);
  Image1.Picture.LoadFromFile(OutputPath);

  if CheckBoxFit.Checked then
  CheckBoxFitChange(Sender);
end;


procedure TForm1.CheckBoxFitChange(Sender: TObject);
var
  SrcBmp, ScaledBmp: TBitmap;
  RatioW, RatioH, ScaleFactor: Double;
  NewWidth, NewHeight: Integer;
begin
  if WorkingBMPPath = '' then Exit;

  if CheckBoxFit.Checked then
  begin
    // 1. Turn off scrollbars while fitting
    ScrollBox1.AutoScroll := False;

    SrcBmp := TBitmap.Create;
    ScaledBmp := TBitmap.Create;
    try
      SrcBmp.LoadFromFile(WorkingBMPPath);

      // 2. Calculate scaling ratios to fit inside ScrollBox client area
      RatioW := ScrollBox1.ClientWidth / SrcBmp.Width;
      RatioH := ScrollBox1.ClientHeight / SrcBmp.Height;

      if RatioW < RatioH then
        ScaleFactor := RatioW
      else
        ScaleFactor := RatioH;

      // Ensure we don't scale up small images if we only want to fit large ones down
      if ScaleFactor > 1.0 then
        ScaleFactor := 1.0;

      NewWidth := Round(SrcBmp.Width * ScaleFactor);
      NewHeight := Round(SrcBmp.Height * ScaleFactor);

      // 3. Create a scaled target bitmap
      ScaledBmp.PixelFormat := pf24bit;
      ScaledBmp.Width := NewWidth;
      ScaledBmp.Height := NewHeight;

      // Draw the original resized into the new bitmap
      ScaledBmp.Canvas.StretchDraw(Rect(0, 0, NewWidth, NewHeight), SrcBmp);

      // 4. Assign the scaled image to Image1
      Image1.Picture.Assign(ScaledBmp);
      Image1.Width := NewWidth;
      Image1.Height := NewHeight;
      Image1.Stretch := False; // Handled manually by our bitmap resizing
      Image1.Proportional := False;

      // 5. Center Image1 inside the ScrollBox viewport
      if ScrollBox1.ClientWidth > Image1.Width then
        Image1.Left := (ScrollBox1.ClientWidth - Image1.Width) div 2
      else
        Image1.Left := 0;

      if ScrollBox1.ClientHeight > Image1.Height then
        Image1.Top := (ScrollBox1.ClientHeight - Image1.Height) div 2
      else
        Image1.Top := 0;

    finally
      SrcBmp.Free;
      ScaledBmp.Free;
    end;
  end
  else
  begin
    // Revert back to native 1:1 pixel size with scrollbars enabled
    Image1.Stretch := False;
    Image1.Proportional := False;
    Image1.Picture.LoadFromFile(WorkingBMPPath);
    Image1.Width := Image1.Picture.Width;
    Image1.Height := Image1.Picture.Height;
    Image1.Left := 0;
    Image1.Top := 0;
    ScrollBox1.AutoScroll := True;
  end;
end;

end.
